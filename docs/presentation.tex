% Slide 2: Executive Summary
\begin{frame}{1. Executive Summary}
    \begin{itemize}
        \item \textbf{Overview:} The \textbf{PFM AI App} is a state-of-the-art, full-stack Personal Finance Manager designed to run in a Windows PowerShell environment.
        \item \textbf{Value Proposition:} Minimizes manual friction using automated categorization, anomaly detection, predictive forecasting, and a natural language chatbot interface.
        \item \textbf{Architecture:} Decoupled three-tier architecture utilizing a \textbf{Python FastAPI} backend, \textbf{React 18 / Vite} frontend, and a \textbf{PostgreSQL} database.
        \item \textbf{Infrastructure:} Fully containerized and orchestrated via Docker, with PowerShell scripting for seamless local developer onboarding.
    \end{itemize}
\end{frame}

% Slide 8: AI Predictive Forecasting Engine
\begin{frame}{7. AI Predictive Expense Forecasting}
    \begin{block}{Ordinary Least Squares (OLS) Linear Regression}
        Predicts next-month category spending using ordinal monthly indexes:
        $$\hat{y}(t+1) = \beta_0 + \beta_1 \times t$$
        where $t \in \{0, 1, ..., n-1\}$, $\beta_0$ is the intercept, and $\beta_1$ is spending velocity.
    \end{block}

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Confidence Score Mapping:}
            $$\text{Confidence} = \min(95, 50 + R^2 \times 45) \to [50, 95]\%$$
            
            \textbf{Trend Assignment:}
            \begin{itemize}
                \item \textbf{Increasing:} $\beta_1 > 0.05 \times \mu_c$
                \item \textbf{Decreasing:} $\beta_1 < -0.05 \times \mu_c$
                \item \textbf{Stable:} otherwise (within $\pm 5\%$)
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Benchmarks (MAPE vs History):}
            \begin{itemize}
                \item 6-11 months history: 18.3\% MAPE
                \item $\ge 12$ months history: \textbf{12.1\% MAPE}
            \end{itemize}
            \begin{block}{Cold Start Fallback}
                For $< 2$ months of history, defaults to the historical mean with $60\%$ confidence and a \texttt{stable} trend.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% Slide 9: Predictive AI Savings Advisor & Actionable Transfers
\begin{frame}{8. Predictive AI Savings Advisor \& Actionable Transfers}
    To elevate savings tracking, we upgraded the engine to perform predictive cash flow audits and interactive actions:
    
    \begin{block}{1. Multi-Factor Cash Flow Runway Auditing}
        Analyzes the last 30 days of transactions (income vs. spent) to calculate your \textbf{net monthly savings rate} and \textbf{daily burn rate}.
        Warns if current bank balance covers $<15$ days of spending.
    \end{block}

    \begin{block}{2. Actionable AI Recommendations (One-Click Application)}
        \begin{itemize}
            \item \textbf{Off-Track Goals:} Automatically calculates required monthly increase to hit target deadlines, allowing adjustment with one click.
            \item \textbf{Surplus Sweep:} Suggests sweeping 30\% of net monthly surplus cash flow directly into priority savings goals.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{12. Core Integrations \& Advanced Utilities}

            \begin{itemize}
                \item Handled by the `ConnectionManager` class in \texttt{notifications\_api.py}.

% Slide 14: Deployment & Local Setup Workflow
\begin{frame}{13. Deployment \& Local Setup Workflow}
    The project is optimized for developers and system deployers using custom scripts:
    
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{PowerShell Automation (Windows)}
                \begin{itemize}
                    \item \textbf{\texttt{setup.ps1}:} Detects prerequisites (Python, Node, Docker), creates backend \texttt{.venv}, installs NPM modules, and orchestrates containers.

                    \item \textbf{\texttt{start-frontend.ps1}:} Boots Vite server.
                    \item \textbf{\texttt{train-model.ps1}:} Retrains ML models.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Docker Orchestration}
                Configured via \texttt{docker-compose.yml} to coordinate:
                \begin{itemize}
                    \item \textbf{\texttt{db}:} PostgreSQL 17 database.
                    \item \textbf{\texttt{redis}:} Caching and Pub/Sub channel.
                    \item \textbf{\texttt{pgadmin}:} Visual database admin GUI at port 5050.
